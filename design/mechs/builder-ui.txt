const { catalog, buildPrompt } = globalThis.mechDesign;
const variant=document.querySelector('#variant'), mounts=document.querySelector('#mounts'), output=document.querySelector('#output'), status=document.querySelector('#status');
for(const f of catalog.frames)variant.add(new Option(f.name,f.id));variant.value='warden';
const getFrame=()=>catalog.frames.find(f=>f.id===variant.value);
const baseline=()=>globalThis.approvedMechBaselines.find(b=>b.chassisId===getFrame().chassisId);
const request=()=>({schemaVersion:1,mode:document.querySelector("#mode").value,...(document.querySelector("#mode").value==="variant"?{baselineId:baseline()?.id}:{}),variantId:variant.value,background:document.querySelector('#background').value,details:document.querySelector('#details').value,loadout:[...mounts.children].map(row=>({weaponId:row.querySelector('select').value||null,rounds:Number(row.querySelector('input').value)}))});
function update(){try{output.value=buildPrompt(catalog,request(),baseline());status.textContent='Valid configuration · '+getFrame().mounts.length+' mounts';}catch(e){output.value='';status.textContent=e.message;}}
function reset(){mounts.replaceChildren();const f=getFrame();for(const [i,m] of f.mounts.entries()){const row=document.createElement('label');row.textContent=`${i+1}. ${m.name} · ${m.family}`;const select=document.createElement('select');select.add(new Option('Empty interface',''));for(const id of m.weapons){const w=catalog.weapons.find(w=>w.id===id);select.add(new Option(w.name+' · '+w.type,id));}select.value=document.querySelector('#preset').value==='starter'?(i===0?f.starter:''):m.weapons[0];const input=document.createElement('input');input.type='number';input.min='0';input.step='1';input.setAttribute('aria-label','Carried rounds for '+m.name);const sync=()=>{const w=catalog.weapons.find(w=>w.id===select.value);input.max=String(w?.ammunition?.capacity??0);input.value=String(w?.ammunition?.defaultLoad??0);input.disabled=!w?.ammunition;update();};row.append(select,document.createTextNode('Carried rounds'),input);mounts.append(row);select.onchange=sync;input.oninput=update;sync();}update();}
document.querySelector('#mode').onchange=update;variant.onchange=reset;document.querySelector('#preset').onchange=reset;document.querySelector('#background').onchange=update;document.querySelector('#details').oninput=update;
function download(name,text){const a=document.createElement('a'),url=URL.createObjectURL(new Blob([text],{type:'text/plain'}));a.href=url;a.download=name;a.click();setTimeout(()=>URL.revokeObjectURL(url),1000);}
document.querySelector('#copy').onclick=async()=>{if(!output.value)return;try{await navigator.clipboard.writeText(output.value);status.textContent='Prompt copied.';}catch{output.focus();output.select();status.textContent='Clipboard unavailable. Selected the prompt; use your system Copy command.';}};
document.querySelector('#save').onclick=()=>{if(output.value)download(variant.value+'-prompt.txt',output.value);};
document.querySelector('#json').onclick=()=>{if(output.value)download(variant.value+'-request.json',JSON.stringify(request(),null,2));};reset();
